# kitty (with vim keybinds)
# kitty_mod for everything = control+shift
# v = paste_from_clipboard
# s paste_from_selection (middle click)
#
# Scroll
# j = up 1
# k = down 1
# up = up page
# down = down page
# / = search
# g = show_last_command_output with less
#
# tab
# t = new tab
# w = close window
# right or l or ctrl+tab = tab to right
# left or h or ctrl+shift+tab = tab to right
# . = move tab right
# , = move tab left
# 1-10 = tab 1-10
# kitty_mod+alt+t = set_tab_title
#
# window (Fat)
# enter new_window
# ] or } = next_window
# [ or { = previous_window
# ` = move_window_to_top
# ctrl+alt+ (1-9) = (1-9) window
# f8 swap_with_window
#
# font
# equal or plus = font_size +2
# minus = font_size -2.0
# backspace = reset font_size
#
# Miscellaneous
# f1 = show documentation (cool file website)
# f3 command palette (show keybinds)
# f11 or kitty_mod+f11 = toggle_fullscreen
# u = unicode input (😀)
